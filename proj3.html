MODULE_02   TB TOPOLOGY

TESTBENCH ARCHITECTURE

Class-based environment communicating purely through mailbox handles — no ports, no interfaces between components. mem_env forks the agent and scoreboard; mem_agnt forks the generator, BFM, monitor and coverage collector.

MODULE_03   OPERATION

VERIFICATION FLOW

⌘
GENERATE
  • mem_gen selects testcase from common::testcase
  • tx.randomize() with inline constraints
  • For read-after-write, address is captured in a mem_tx temp[$] queue
  • Pushed to gen2bfm mailbox
▶
DRIVE
  • mem_bfm gets tx from gen2bfm
  • Drives pins via vif.bfm_cb clocking block
  • Asserts valid_i, waits for ready_o
  • Resets signals, increments common::bfm_count
◈
SAMPLE
  • mem_mon is a passive, pin-to-transaction observer
  • Samples on vif.mon_cb when valid_i & ready_o
  • Rebuilds mem_tx from bus activity
  • Broadcasts to mon2scb and mon2cov
✓
CHECK
  • mem_scb keeps a shadow bit [width-1:0] mem[*]
  • On write: updates shadow entry at addr_i
  • On read: compares DUT rdata_o to shadow value
  • Increments common::matching / mismatch
⬡
COVER
  • mem_cov owns a covergroup cg
  • Coverpoints on wr_rd_i and addr_i
  • Cross bin WR_RD_X_ADDR
  • cg.sample() called per transaction from mon2cov
MODULE_04   TESTPLAN

DIRECTED TESTCASES

TC_01
1_WRITE
Randomizes and issues a single wr_rd_i = 1 transaction. Baseline sanity check for a lone write handshake.
Sequence: randomize(WRITE) ×1
TC_02
N_WRITE
Loops common::N times, each a randomized write. Exercises the generator/BFM handshake loop under repetition.
Sequence: randomize(WRITE) ×N
TC_03
1_WRITE_1_READ
Writes to a random address, pushes it onto a queue, then issues a read constrained to that same address — a directed read-after-write check.
Sequence: write(addr=A) → read(addr=A)
TC_04
N_WRITE_N_READ
Default testcase. N writes to a queue of addresses, then N reads popped back in reverse order — full read-after-write coverage across the address space.
Sequence: write(addr) ×N → read(addr) ×N
⚙ Selected via common::testcase (default "N_WRITE_N_READ") and common::N (default 5) in mem_common.sv — change the static string to switch testcases, no plusargs used in this version.
MODULE_05   PROTOCOL CHECKS

SVA ASSERTIONS

Six immediate protocol properties live directly inside mem_intf.sv, firing on every clock regardless of which testcase is active.

HANDSHAKEp1
valid_i == 1 |=> ready_o == 1
Every valid request must be acknowledged the next cycle.
WR_RDp2
(valid_i && ready_o) |-> !$isunknown(wr_rd_i)
Direction must be known during an accepted transfer.
ADDR_1p3
wr_rd_i == 1 |-> !$isunknown(addr_i)
Address must be known on every write.
WDATAp4
wr_rd_i == 1 |-> !$isunknown(wdata_i)
Write data must be known on every write.
ADDR_2p5
wr_rd_i == 0 |-> !$isunknown(addr_i)
Address must be known on every read.
RDATAp6
wr_rd_i == 0 |-> !$isunknown(rdata_o)
Read data must resolve to a known value.

module memory_dut #(parameter depth = 16, width = 4,
                     addr_width = $clog2(depth))
  (clk_i,rst_i,wr_rd_i,wdata_i,rdata_o,
   addr_i,valid_i,ready_o);

input clk_i,rst_i,wr_rd_i,valid_i;
input [width-1:0] wdata_i;
input [addr_width-1:0] addr_i;
output reg [width-1:0] rdata_o;
output reg ready_o;

reg [width-1:0] mem [depth-1:0];
integer i;

always@(posedge clk_i or posedge rst_i) begin
  if(rst_i) begin
    ready_o <= 0;
    rdata_o <= 0;
    for(i=0; i<depth; i=i+1) begin
      mem[i] <= 0;
    end
  end
  else begin
    if(valid_i) begin
      ready_o <= 1;
      if(wr_rd_i) mem[addr_i] <= wdata_i;
      else rdata_o <= mem[addr_i];
    end
    else ready_o <= 0;
  end
end

endmodule
interface mem_intf(input clk,rst);

logic wr_rd_i;
logic valid_i;
logic [`width-1:0] wdata_i;
logic [`addr_width-1:0] addr_i;
logic [`width-1:0] rdata_o;
logic ready_o;

// Monitor-side clocking block (sampling)
clocking mon_cb @(posedge clk);
  input wr_rd_i;
  input #1 valid_i;
  input wdata_i;
  input addr_i;
  input rdata_o;
  input #1 ready_o;
endclocking

// BFM-side clocking block (driving)
clocking bfm_cb @(posedge clk);
  input #1 ready_o;
  input rdata_o;
  output wr_rd_i;
  output valid_i;
  output wdata_i;
  output addr_i;
endclocking

// Protocol assertions (see MODULE_05)
property p1; @(posedge clk) (valid_i == 1) |=> (ready_o == 1); endproperty
property p2; @(posedge clk) ((valid_i==1) && (ready_o==1)) |-> !($isunknown(wr_rd_i)); endproperty
property p3; @(posedge clk) (wr_rd_i==1) |-> !($isunknown(addr_i)); endproperty
property p4; @(posedge clk) (wr_rd_i==1) |-> !($isunknown(wdata_i)); endproperty
property p5; @(posedge clk) (wr_rd_i==0) |-> !($isunknown(addr_i)); endproperty
property p6; @(posedge clk) (wr_rd_i==0) |-> !($isunknown(rdata_o)); endproperty

HANDSHAKE: assert property(p1) else $error("HANDSHAKE Assertion Failed!");
WR_RD:     assert property(p2) else $error("WR_RD Assertion Failed!");
ADDR_1:    assert property(p3) else $error("ADDR_1 Assertion Failed!");
WDATA:     assert property(p4) else $error("WDATA Assertion Failed!");
ADDR_2:    assert property(p5) else $error("ADDR_2 Assertion Failed!");
RDATA:     assert property(p6) else $error("RDATA Assertion Failed!");

endinterface
class mem_tx;

rand bit [`width-1:0] wdata_i;
rand bit [`addr_width-1:0] addr_i;
rand bit wr_rd_i;
bit [`width-1:0] rdata_o;

constraint c1{ wdata_i inside {[1:256]}; }

function void print(string name = "TX_CLASS");
  $display("---------------- Name: %s | Time: %0t ----------------",name,$time);
  $display("wr_rd_i: %0b",wr_rd_i);
  $display("Address: %0d",addr_i);
  $display("WData: %0d",wdata_i);
  $display("RData: %0d\n",rdata_o);
endfunction

endclass
class mem_gen;

mem_tx tx;
mem_tx temp[$];
mem_tx t;

task run();
  case(common::testcase)
    "1_WRITE": begin
      tx = new();
      assert(tx.randomize() with {tx.wr_rd_i == 1;});
      gen2bfm.put(tx);
    end
    "N_WRITE": begin
      repeat(common::N) begin
        tx = new();
        assert(tx.randomize() with {tx.wr_rd_i == 1;});
        gen2bfm.put(tx);
      end
    end
    "1_WRITE_1_READ": begin
      begin
        tx = new();
        assert(tx.randomize() with {tx.wr_rd_i == 1;});
        temp.push_front(tx);
        gen2bfm.put(tx);
      end
      begin
        tx = new();
        t = temp.pop_back();
        assert(tx.randomize() with {tx.wr_rd_i == 0; tx.addr_i == t.addr_i;});
        gen2bfm.put(tx);
      end
    end
    "N_WRITE_N_READ": begin
      repeat (common::N) begin
        tx = new();
        assert(tx.randomize() with {tx.wr_rd_i == 1;});
        temp.push_front(tx);
        gen2bfm.put(tx);
      end
      repeat (common::N) begin
        tx = new();
        t = temp.pop_back();
        assert(tx.randomize() with {tx.wr_rd_i == 0; tx.addr_i == t.addr_i;});
        gen2bfm.put(tx);
      end
    end
    default: $error("INVALID TESTCASE!");
  endcase
endtask

endclass
class mem_bfm; // tx lvl info to pin lvl info

mem_tx tx;
virtual mem_intf vif;

function new();
  vif = top.pif;
endfunction

task run();
  forever begin
    wait(top.rst == 0);
    gen2bfm.get(tx);
    drive_tx(tx);
  end
endtask

task reset();
  vif.bfm_cb.wr_rd_i <= 0;
  vif.bfm_cb.addr_i  <= 0;
  vif.bfm_cb.wdata_i <= 0;
  vif.bfm_cb.valid_i <= 0;
  @(vif.bfm_cb);
endtask

task drive_tx(mem_tx tx);
  @(vif.bfm_cb);
  vif.bfm_cb.addr_i   <= tx.addr_i;
  vif.bfm_cb.wr_rd_i  <= tx.wr_rd_i;
  if(tx.wr_rd_i == 1) vif.bfm_cb.wdata_i <= tx.wdata_i;
  else vif.bfm_cb.wdata_i <= 0;
  vif.bfm_cb.valid_i <= 1;
  wait(vif.bfm_cb.ready_o == 1);
  vif.bfm_cb.valid_i <= 0;
  reset();
  common::bfm_count++;
endtask

endclass
class mem_mon; // pin lvl to tx lvl

mem_tx tx;
virtual mem_intf vif;

function new();
  vif = top.pif;
endfunction

task run();
  forever begin
    @(vif.mon_cb);
    if((vif.mon_cb.valid_i == 1) && (vif.mon_cb.ready_o == 1)) begin
      tx = new();
      tx.wr_rd_i = vif.mon_cb.wr_rd_i;
      tx.addr_i  = vif.mon_cb.addr_i;
      if(tx.wr_rd_i == 1) begin
        tx.wdata_i = vif.mon_cb.wdata_i;
        mon2scb.put(tx);
        mon2cov.put(tx);
      end
      else begin
        tx.rdata_o = vif.mon_cb.rdata_o;
        mon2scb.put(tx);
        mon2cov.put(tx);
      end
    end
  end
endtask

endclass
class mem_scb;

mem_tx tx;
bit [`width-1:0] mem[*]; // associative array shadow memory

task run();
  forever begin
    mon2scb.get(tx);
    if(tx.wr_rd_i == 1)
      mem[tx.addr_i] = tx.wdata_i;
    else begin
      if(tx.rdata_o == mem[tx.addr_i]) common::matching++;
      else common::mismatch++;
    end
  end
endtask

endclass
class mem_cov;

mem_tx tx;

covergroup cg;
  WR_RD: coverpoint tx.wr_rd_i{
    bins WRITE = {1'b1};
    bins READ  = {1'b0};
  }
  ADDR: coverpoint tx.addr_i{
    bins LOWER_ADDR     = {[0:3]};
    bins LOWER_MID_ADDR = {[4:7]};
    bins MID_ADDR       = {[8:11]};
    bins UPPER_ADDR     = {[12:15]};
  }
  WR_RD_X_ADDR: cross WR_RD, ADDR{}
endgroup

function new();
  cg = new();
endfunction

task run();
  forever begin
    mon2cov.get(tx);
    tx.print("COV_TX");
    cg.sample();
  end
endtask

endclass

module top;

bit clk,rst;
mem_env env;
mem_intf pif(clk,rst);

memory_dut #(.depth(`depth), .width(`width), .addr_width(`addr_width)) dut
(
  .clk_i(clk), .rst_i(rst),
  .wr_rd_i(pif.wr_rd_i), .wdata_i(pif.wdata_i),
  .rdata_o(pif.rdata_o), .addr_i(pif.addr_i),
  .valid_i(pif.valid_i), .ready_o(pif.ready_o)
);

always #5 clk=~clk;

initial begin
  clk = 0; rst = 1;
  pif.valid_i = 0; pif.wr_rd_i = 0;
  pif.addr_i = 0; pif.wdata_i = 0;
  repeat(2) @(posedge clk);
  rst = 0;
end

initial begin
  env = new();
  env.run();
end

initial #1000 $finish();

initial begin
  wait(common::bfm_count == common::N*2);
  if(common::matching != 0 && common::matching == common::N && common::mismatch == 0)
    $display("TESTCASE PASSED!...");
  else $display("TESTCASE FAILED!...");
end

endmodule
